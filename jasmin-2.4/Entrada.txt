DOUBLE a = 0;
DOUBLE b = 1;
DOUBLE e = 2.718281828459045;
INT n = 1000;
DOUBLE h = (b - a) / n;
DOUBLE soma = e^a * a^2 + e^b * b^2;
DOUBLE x = 0;

for (INT i = 1; i < n; i++) {
    x = a + i * h;
    soma = soma + 2 * (e^x * x^2);
}

DOUBLE resultado = (h / 2) * soma;

/*' 

    UTILIZANDO EXPRESSAO TERNARIA

'*/

STRING mensagem := (resultado>0) ? 'Resultado da integracao ' : 'ERRO ';
